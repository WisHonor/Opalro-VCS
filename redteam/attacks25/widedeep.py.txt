#!/usr/bin/env python3
"""A Git repository of two commits whose comparison holds a long path for every folder it has
still to compare: the top tree reaches, through levels named with `ln` letters each, a folder F
about `depth` bytes down, of `n` folders s0000.. . In the first commit each of those holds the
file x as the blob A, in the second as the blob B, so all n differ, and the comparison stacks
all n paths, each about `depth` bytes, before it compares any of them.
Usage: widedeep.py <dir> <n> <depth>"""
import hashlib, os, subprocess, sys, zlib

root, n, depth = sys.argv[1], int(sys.argv[2]), int(sys.argv[3])
git = os.path.join(root, ".git")
subprocess.run(["git", "init", "-q", "-b", "main", root], check=True)
objects = os.path.join(git, "objects")


def put(kind, data):
    raw = b"%s %d\0" % (kind, len(data)) + data
    oid = hashlib.sha1(raw).digest()
    hexid = oid.hex()
    folder = os.path.join(objects, hexid[:2])
    os.makedirs(folder, exist_ok=True)
    with open(os.path.join(folder, hexid[2:]), "wb") as f:
        f.write(zlib.compress(raw, 9))
    return oid


def name36(i, width):
    digits = "0123456789abcdefghijklmnopqrstuvwxyz"
    s = ""
    for _ in range(width):
        s = digits[i % 36] + s
        i //= 36
    return s.encode()


# Level names: as many 765-letter ones as fit, then one for the rest, so F's path is `depth`.
levels, left = [], depth
while left > 0:
    size = min(765, left)
    if left - size == 1:  # never leave a single byte for a name and its slash
        size -= 2
    levels.append(size)
    left -= size + 1
print("levels", levels, "path", sum(levels) + len(levels) - 1)


def commit(blob, parent, i):
    e = put(b"tree", b"100644 x\0" + blob)
    f = put(b"tree", b"".join(b"40000 s" + name36(j, 4) + b"\0" + e for j in range(n)))
    t = f
    for level, size in reversed(list(enumerate(levels))):
        name = (b"%d" % level) + b"a" * (size - 1)
        t = put(b"tree", b"40000 " + name + b"\0" + t)
    text = b"tree %s\n" % t.hex().encode()
    if parent:
        text += b"parent %s\n" % parent.hex().encode()
    text += b"author T <t@e> %d +0000\ncommitter T <t@e> %d +0000\n\nc%d\n" % (
        1000000000 + i, 1000000000 + i, i)
    return put(b"commit", text)


a = put(b"blob", b"A\n")
b = put(b"blob", b"B\n")
tip = commit(b, commit(a, None, 0), 1)
with open(os.path.join(git, "refs", "heads", "main"), "w") as f:
    f.write(tip.hex() + "\n")
print("written", root)
