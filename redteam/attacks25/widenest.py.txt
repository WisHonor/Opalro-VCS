#!/usr/bin/env python3
"""widedeep.py with the wide folders nested: the folder about `depth` bytes down holds `n`
folders s0000.. that each hold the file x (blob A in the first commit, B in the second), and,
sorted last, `z`, which holds the same again, `nest` levels in all. The comparison lists the n
paths of one level, then goes into `z` first and lists n more on top: about nest * n paths of
4 KB each at once.
Usage: widenest.py <dir> <n> <nest> <depth>"""
import hashlib, os, subprocess, sys, zlib

root, n, nest, depth = sys.argv[1], int(sys.argv[2]), int(sys.argv[3]), int(sys.argv[4])
git = os.path.join(root, ".git")
subprocess.run(["git", "init", "-q", "-b", "main", root], check=True)
objects = os.path.join(git, "objects")


def put(kind, data):
    raw = b"%s %d\0" % (kind, len(data)) + data
    oid = hashlib.sha1(raw).digest()
    hexid = oid.hex()
    folder = os.path.join(objects, hexid[:2])
    os.makedirs(folder, exist_ok=True)
    with open(os.path.join(folder, hexid[2:]), "wb") as f:
        f.write(zlib.compress(raw, 9))
    return oid


def name36(i, width):
    digits = "0123456789abcdefghijklmnopqrstuvwxyz"
    s = ""
    for _ in range(width):
        s = digits[i % 36] + s
        i //= 36
    return s.encode()


levels, left = [], depth
while left > 0:
    size = min(765, left)
    if left - size == 1:
        size -= 2
    levels.append(size)
    left -= size + 1


def commit(blob, parent, i):
    e = put(b"tree", b"100644 x\0" + blob)
    wide = b"".join(b"40000 s" + name36(j, 4) + b"\0" + e for j in range(n))
    f = put(b"tree", wide)
    for _ in range(nest - 1):
        f = put(b"tree", wide + b"40000 z\0" + f)
    t = f
    for level, size in reversed(list(enumerate(levels))):
        t = put(b"tree", b"40000 " + (b"%d" % level) + b"a" * (size - 1) + b"\0" + t)
    text = b"tree %s\n" % t.hex().encode()
    if parent:
        text += b"parent %s\n" % parent.hex().encode()
    text += b"author T <t@e> %d +0000\ncommitter T <t@e> %d +0000\n\nc%d\n" % (
        1000000000 + i, 1000000000 + i, i)
    return put(b"commit", text)


tip = commit(put(b"blob", b"B\n"), commit(put(b"blob", b"A\n"), None, 0), 1)
with open(os.path.join(git, "refs", "heads", "main"), "w") as f:
    f.write(tip.hex() + "\n")
print("written", root)
