//! RT18 probes: print what each command says.
#![allow(clippy::unwrap_used, clippy::expect_used, clippy::panic)]

use opal_core::diary::{Action, DiaryEntry, State};
use opal_core::id::Id;
use opal_core::repo::Repository;

mod support;
use support::{Env, text};

fn run(env: &Env, args: &[&str]) -> (Option<i32>, String, String) {
    let out = env.opalro().args(args).output().unwrap();
    let r = (out.status.code(), text(&out.stdout), text(&out.stderr));
    eprintln!("$ opalro {}\n  code {:?}\n  out: {}\n  err: {}", args.join(" "), r.0, r.1, r.2);
    r
}

fn plant_state(env: &Env, change: impl FnOnce(&mut State)) {
    let mut repo = Repository::open(env.root()).unwrap();
    let lock = repo.lock_for_change("tag", &mut |_| {}).unwrap();
    let (head, entry) = repo.current().unwrap();
    let mut state = entry.state.clone();
    change(&mut state);
    let packs = repo.new_packs().unwrap();
    let diary = DiaryEntry {
        parent: Some(head),
        action: Action::Command,
        command: "tag".into(),
        who: entry.who,
        opal_version: "0.1.0".into(),
        state,
        folder_change: None,
    };
    repo.commit(&lock, packs, &diary).unwrap();
}

#[test]
fn rt18_p1_server_line_ahead_and_tag() {
    let env = Env::new();
    let one = env.commit("a.txt", "one", "one");
    let two = env.commit("a.txt", "two", "two");
    env.git(&["update-ref", "refs/remotes/origin/main", &two]);
    env.git(&["reset", "-q", "--hard", &one]);
    env.git(&["tag", "main@origin", &one]);
    env.import();
    let (repo, state) = env.state();
    let server = repo.server_lines(&state).unwrap().servers["origin"]["main"];
    drop(repo);
    run(&env, &["goto", "main@origin"]);
    run(&env, &["goto", &server.to_hex()[..12]]);
    run(&env, &["goto", &server.to_hex()]);
    run(&env, &["show", &server.to_hex()[..12]]);
}

#[test]
fn rt18_p2_line_and_tag_off_lines() {
    let env = Env::new();
    env.commit("a.txt", "one", "one");
    env.git(&["checkout", "-q", "-b", "tmp"]);
    env.commit("a.txt", "two", "two");
    env.git(&["tag", "v1"]);
    env.git(&["checkout", "-q", "main"]);
    env.git(&["branch", "-q", "-D", "tmp"]);
    env.git(&["branch", "v1"]);
    env.import();
    let (repo, state) = env.state();
    let tag = repo.tag_list(&state).unwrap().tags["v1"].save;
    drop(repo);
    run(&env, &["goto", "v1"]);
    run(&env, &["goto", &tag.to_hex()[..12]]);
    run(&env, &["goto", &tag.to_hex()]);
    run(&env, &["tag"]);
    run(&env, &["line", "rename", "v1", "v1-branch"]);
    run(&env, &["show", "v1"]);
}

fn project() -> Env {
    let env = Env::empty();
    env.opalro().args(["start", "--preset", "custom"]).assert().success();
    env.overwrite("a.txt", "a");
    env.opalro().args(["save", "A"]).assert().success();
    env
}

#[test]
fn rt18_p3_damaged_tag_list() {
    let env = project();
    env.opalro().args(["line", "new", "feat"]).assert().success();
    env.overwrite("a.txt", "f");
    env.opalro().args(["save", "F"]).assert().success();
    let bogus = Id::from_hex(&"6".repeat(64)).unwrap();
    plant_state(&env, |s| s.tags = Some(bogus));
    run(&env, &["goto", "main"]);
    run(&env, &["history", "--json"]);
    run(&env, &["save", "x"]);
    run(&env, &["tag", "--delete", "x", "--yes"]);
    run(&env, &["line", "delete", "feat", "--yes"]);
    run(&env, &["undo"]);
    run(&env, &["goto", "main"]);
}

#[test]
fn rt18_p4_holder_with_missing_parent() {
    let env = project();
    env.opalro().args(["line", "new", "feat"]).assert().success();
    env.overwrite("a.txt", "f");
    env.opalro().args(["save", "F"]).assert().success();
    env.opalro().args(["goto", "main"]).assert().success();
    // A save whose parent isn't in the store, tagged.
    let (repo, state) = env.state();
    let tip = repo.store().save(&state.lines["main"], false).unwrap();
    drop(repo);
    let missing = Id::from_hex(&"5".repeat(64)).unwrap();
    let mut planted = None;
    support::add_pack(env.root(), |pack| {
        let save = opal_core::save::Save {
            parents: vec![missing],
            change_id: opal_core::id::ChangeId::random().unwrap(),
            message: "orphan".into(),
            ..tip
        };
        planted = Some(
            pack.add_record(opal_core::pack::ObjectKind::Save, &save.encode())
                .unwrap(),
        );
    });
    env.plant_tags(["ghost".to_owned()].into_iter(), planted.unwrap());
    run(&env, &["line", "delete", "feat", "--yes"]);
    run(&env, &["line", "delete", "feat", "--yes", "--json"]);
}

#[test]
fn rt18_p5_show_tag_on_other_line_hint() {
    let env = project();
    env.opalro().args(["line", "new", "feat"]).assert().success();
    env.overwrite("a.txt", "f");
    env.opalro().args(["save", "F"]).assert().success();
    env.opalro().args(["tag", "rel"]).assert().success();
    env.opalro().args(["goto", "main"]).assert().success();
    let (_, out, _) = run(&env, &["show", "rel"]);
    let _ = out;
}

#[test]
fn rt18_p6_tag_named_main_elsewhere() {
    let env = project();
    let (repo, state) = env.state();
    let first = repo.store().save(&state.lines["main"], false).unwrap().parents[0];
    drop(repo);
    env.plant_tags(["main".to_owned()].into_iter(), first);
    run(&env, &["goto", "main"]);
    run(&env, &["show", "main"]);
    run(&env, &["history"]);
    run(&env, &["save", "x"]);
    run(&env, &["line"]);
    run(&env, &["tag"]);
}
