struct Rng(u64);
impl Rng {
    fn next(&mut self) -> u64 {
        self.0 ^= self.0 << 13;
        self.0 ^= self.0 >> 7;
        self.0 ^= self.0 << 17;
        self.0
    }
    fn below(&mut self, n: usize) -> usize {
        (self.next() % n as u64) as usize
    }
}

fn reaches(saves: &[(Id, Vec<Id>)], from: Id, to: Id) -> bool {
    let parents: BTreeMap<Id, Vec<Id>> = saves.iter().cloned().collect();
    let mut seen = HashSet::new();
    let mut next = vec![from];
    while let Some(id) = next.pop() {
        if id == to {
            return true;
        }
        if seen.insert(id) {
            next.extend(parents.get(&id).cloned().unwrap_or_default());
        }
    }
    false
}

fn run(env: &Env, args: &[&str]) -> (Option<i32>, String, String) {
    let out = env.opalro().args(args).output().unwrap();
    (out.status.code(), text(&out.stdout), text(&out.stderr))
}

fn key(l: &[u8]) -> &[u8] {
    let l = l.strip_suffix(b"\n").unwrap_or(l);
    l.strip_suffix(b"\r").unwrap_or(l)
}

/// Every line `who --json` names must be held by the named save at the named path, at the line
/// number the walk followed it to being equal content-wise.
fn check_who(env: &Env, file: &str, extra: &[&str], body: &[u8], deadline: Option<&str>) -> usize {
    let mut cmd = env.opalro();
    cmd.args(["who", file, "--json"]).args(extra);
    if let Some(d) = deadline {
        cmd.env("OPALRO_DIFF_DEADLINE_MS", d);
    }
    let out = cmd.output().unwrap();
    assert_eq!(out.status.code(), Some(0), "{file} {extra:?} {}", text(&out.stderr));
    let doc: serde_json::Value = serde_json::from_slice(&out.stdout).unwrap();
    let lines: Vec<&[u8]> = body.split_inclusive(|b| *b == b'\n').collect();
    let mut checked = 0;
    for r in doc["runs"].as_array().unwrap() {
        let Some(save) = r["save"].as_str() else {
            continue;
        };
        let id = Id::from_hex(save).unwrap();
        let path = r["was"].as_str().unwrap_or(doc["path"].as_str().unwrap());
        let files = support::saved_files(env, id);
        let held = files
            .get(path)
            .unwrap_or_else(|| panic!("{file}: {save} has no {path}: {doc}"));
        let keys: Vec<&[u8]> = held.split_inclusive(|b| *b == b'\n').map(key).collect();
        let first = r["first"].as_u64().unwrap() as usize;
        for n in first..first + r["lines"].as_u64().unwrap() as usize {
            let line = key(lines[n - 1]);
            assert!(
                keys.contains(&line),
                "{file} {extra:?} {deadline:?}: line {n} {:?} named for {save} at {path}, which lacks it",
                String::from_utf8_lossy(line)
            );
            checked += 1;
        }
    }
    checked
}

/// Random edits, moves and swaps of three files on two lines with combines; `who` on each.
#[test]
fn rt24_who_moves_random() {
    let names = ["a.txt", "b.txt", "c.txt", "d/e.txt"];
    let mut total = 0;
    for seed in 1..=12u64 {
        let mut rng = Rng(0x2545_f491_4f6c_dd1d ^ seed.wrapping_mul(0x9e37_79b9));
        let env = Env::empty();
        assert_eq!(run(&env, &["start", "--preset", "custom"]).0, Some(0));
        std::fs::create_dir_all(env.root().join("d")).unwrap();
        for (i, n) in names.iter().enumerate().take(2) {
            env.overwrite(n, &format!("{i}0\n{i}1\n{i}2\nsame\n"));
        }
        assert_eq!(run(&env, &["save", "First"]).0, Some(0));
        assert_eq!(run(&env, &["line", "new", "side"]).0, Some(0));
        let lines = ["main", "side"];
        let mut on = 1;
        let mut tips = Vec::new();
        for step in 0..30 {
            let present: Vec<&str> = names
                .iter()
                .copied()
                .filter(|n| env.root().join(n).exists())
                .collect();
            let absent: Vec<&str> = names
                .iter()
                .copied()
                .filter(|n| !env.root().join(n).exists())
                .collect();
            match rng.below(8) {
                0 => {
                    on = 1 - on;
                    let (c, _, e) = run(&env, &["goto", lines[on]]);
                    assert_eq!(c, Some(0), "{e}");
                }
                1 => {
                    let (c, o, e) = run(&env, &["combine", lines[1 - on]]);
                    if c != Some(0) || o.contains("conflict") || e.contains("conflict") {
                        for n in names {
                            let _ = run(&env, &["resolve", "--keep", "mine", n]);
                        }
                        let (c2, _, e2) = run(&env, &["status"]);
                        eprintln!("seed {seed} step {step}: combine {c:?}; status {c2:?} {e2}");
                    }
                }
                2 if !present.is_empty() && !absent.is_empty() => {
                    let f = present[rng.below(present.len())];
                    let t = absent[rng.below(absent.len())];
                    std::fs::rename(env.root().join(f), env.root().join(t)).unwrap();
                    let _ = run(&env, &["save", &format!("move {f} {t} {step}")]);
                }
                3 if present.len() >= 2 => {
                    let f = present[0];
                    let t = present[1 + rng.below(present.len() - 1)];
                    let tmp = env.root().join("swap.tmp");
                    std::fs::rename(env.root().join(f), &tmp).unwrap();
                    std::fs::rename(env.root().join(t), env.root().join(f)).unwrap();
                    std::fs::rename(&tmp, env.root().join(t)).unwrap();
                    let _ = run(&env, &["save", &format!("swap {f} {t} {step}")]);
                }
                _ if !present.is_empty() => {
                    let f = present[rng.below(present.len())];
                    let old = std::fs::read(env.root().join(f)).unwrap();
                    let mut ls: Vec<Vec<u8>> = old
                        .split_inclusive(|b| *b == b'\n')
                        .map(<[u8]>::to_vec)
                        .collect();
                    let at = rng.below(ls.len() + 1);
                    let t = format!("{step}-{}\n", rng.below(3)).into_bytes();
                    match rng.below(3) {
                        0 if at < ls.len() => {
                            ls.remove(at);
                        }
                        1 if at < ls.len() => ls[at] = t,
                        _ => ls.insert(at, t),
                    }
                    std::fs::write(env.root().join(f), ls.concat()).unwrap();
                    let _ = run(&env, &["save", &format!("edit {f} {step}")]);
                }
                _ => {}
            }
            if step % 10 == 9 {
                tips.push(support::main_tip(&env));
            }
        }
        for d in [None, Some("0")] {
            for n in names {
                if let Ok(body) = std::fs::read(env.root().join(n)) {
                    total += check_who(&env, n, &[], &body, d);
                }
                for t in &tips {
                    let files = support::saved_files(&env, *t);
                    if let Some(body) = files.get(n) {
                        total += check_who(&env, n, &["--save", &t.to_hex()], body, d);
                    }
                }
            }
        }
    }
    eprintln!("checked {total} lines");
}

/// `combine` and `resolve` started together on a project with a fixed conflict; the fix must
/// survive in a save or the folder, and `verify` must pass.
#[test]
fn rt24_combine_and_resolve_together() {
    let mut outcomes = BTreeMap::new();
    for round in 0..16 {
        let env = Env::empty();
        assert_eq!(run(&env, &["start", "--preset", "custom"]).0, Some(0));
        env.overwrite("a.txt", "one\ntwo\nthree\n");
        env.overwrite("b.txt", "bee\n");
        assert_eq!(run(&env, &["save", "First"]).0, Some(0));
        assert_eq!(run(&env, &["line", "new", "side"]).0, Some(0));
        env.overwrite("a.txt", "one\nSIDE\nthree\n");
        assert_eq!(run(&env, &["save", "Side"]).0, Some(0));
        assert_eq!(run(&env, &["goto", "main"]).0, Some(0));
        assert_eq!(run(&env, &["line", "new", "third"]).0, Some(0));
        env.overwrite("b.txt", "bee\nthird\n");
        assert_eq!(run(&env, &["save", "Third"]).0, Some(0));
        assert_eq!(run(&env, &["goto", "main"]).0, Some(0));
        env.overwrite("a.txt", "one\nMAIN\nthree\n");
        assert_eq!(run(&env, &["save", "Main"]).0, Some(0));
        let (c, o, e) = run(&env, &["combine", "side"]);
        assert!(o.contains("conflict") || e.contains("conflict"), "{c:?} {o} {e}");
        env.overwrite("a.txt", "one\nFIXED\nthree\n");
        let mut a = env.opal_process_in(env.root());
        a.args(["resolve"]);
        let mut b = env.opal_process_in(env.root());
        b.args(["combine", "third"]);
        if round % 2 == 1 {
            std::mem::swap(&mut a, &mut b);
        }
        let ca = a.stdout(std::process::Stdio::piped()).stderr(std::process::Stdio::piped()).spawn().unwrap();
        let cb = b.stdout(std::process::Stdio::piped()).stderr(std::process::Stdio::piped()).spawn().unwrap();
        let oa = ca.wait_with_output().unwrap();
        let ob = cb.wait_with_output().unwrap();
        let key = format!(
            "{:?}/{:?}: {} | {}",
            oa.status.code(),
            ob.status.code(),
            text(&oa.stderr).lines().next().unwrap_or(""),
            text(&ob.stderr).lines().next().unwrap_or("")
        );
        let (vc, vo, ve) = run(&env, &["verify"]);
        assert_eq!(vc, Some(0), "round {round}: verify {vo} {ve}");
        let folder = std::fs::read_to_string(env.root().join("a.txt")).unwrap();
        let tip = main_tip(&env);
        let saved = support::saved_files(&env, tip);
        let in_save = saved.get("a.txt").map(|b| String::from_utf8_lossy(b).into_owned());
        let (_, st, _) = run(&env, &["status"]);
        assert!(
            folder.contains("FIXED") || in_save.as_deref().is_some_and(|s| s.contains("FIXED")),
            "round {round}: the fix is gone: folder {folder:?} save {in_save:?}\n{key}\n{st}"
        );
        *outcomes.entry(key).or_insert(0) += 1;
    }
    eprintln!("{outcomes:#?}");
}

/// Every command run with `.opalro/tmp` that can't be written: what it says, and whether it
/// changed anything.
#[cfg(unix)]
#[test]
fn rt24_tmp_unwritable() {
    use std::os::unix::fs::PermissionsExt;
    let commands: &[&[&str]] = &[
        &["status"],
        &["status", "a.txt"],
        &["changes"],
        &["changes", "a.txt"],
        &["who", "a.txt"],
        &["save", "S"],
        &["save", "S", "a.txt"],
        &["save", "--amend", "S"],
        &["goto", "side"],
        &["goto", "side", "--force"],
        &["combine", "side"],
        &["discard", "a.txt"],
        &["undo"],
        &["line", "new", "other"],
        &["tag", "v1.0"],
        &["shelve"],
        &["verify"],
        &["history"],
        &["move", "a.txt", "c.txt"],
        &["forget", "a.txt"],
        &["ignore", "a.txt"],
    ];
    for args in commands {
        let env = Env::empty();
        assert_eq!(run(&env, &["start", "--preset", "custom"]).0, Some(0));
        env.overwrite("a.txt", "one\ntwo\n");
        env.overwrite("b.txt", "bee\n");
        assert_eq!(run(&env, &["save", "First"]).0, Some(0));
        assert_eq!(run(&env, &["line", "new", "side"]).0, Some(0));
        env.overwrite("b.txt", "side\n");
        assert_eq!(run(&env, &["save", "Side"]).0, Some(0));
        assert_eq!(run(&env, &["goto", "main"]).0, Some(0));
        env.overwrite("a.txt", "one\nunsaved\n");
        let tmp = env.root().join(".opalro").join("tmp");
        let before = (
            support::entries(env.root()),
            support::entries(&env.root().join(".opalro")),
        );
        std::fs::set_permissions(&tmp, std::fs::Permissions::from_mode(0o555)).unwrap();
        let (c, o, e) = run(&env, args);
        std::fs::set_permissions(&tmp, std::fs::Permissions::from_mode(0o755)).unwrap();
        let after = (
            support::entries(env.root()),
            support::entries(&env.root().join(".opalro")),
        );
        let changed = before != after;
        let first = e.lines().next().unwrap_or("").to_owned();
        eprintln!("{args:?}: {c:?} changed {changed}: {first} | {}", o.lines().next().unwrap_or(""));
        if c != Some(0) && changed {
            let b: BTreeMap<_, _> = before.1.iter().collect();
            let a: BTreeMap<_, _> = after.1.iter().collect();
            let gone: Vec<_> = b.keys().filter(|k| !a.contains_key(*k)).collect();
            let new: Vec<_> = a.keys().filter(|k| !b.contains_key(*k)).collect();
            eprintln!("   FAILED BUT CHANGED: gone {gone:?} new {new:?} folder same {}", before.0 == after.0);
            let (vc, vo, ve) = run(&env, &["status"]);
            eprintln!("   status after: {vc:?} {vo} {ve}");
        }
    }
}

/// Unsaved work in a file the other line changed, the file cache in each state, then `goto` and
/// `combine`: the work must survive.
#[test]
fn rt24_cache_states_switch() {
    for variant in 0..7 {
        for cmd in ["goto", "combine"] {
            let env = Env::empty();
            assert_eq!(run(&env, &["start", "--preset", "custom"]).0, Some(0));
            env.overwrite("a.txt", "one\ntwo\n");
            env.overwrite("b.txt", "bee\n");
            assert_eq!(run(&env, &["save", "First"]).0, Some(0));
            assert_eq!(run(&env, &["line", "new", "side"]).0, Some(0));
            env.overwrite("a.txt", "one\nSIDE\n");
            assert_eq!(run(&env, &["save", "Side"]).0, Some(0));
            assert_eq!(run(&env, &["goto", "main"]).0, Some(0));
            // Old enough to be trusted once the cache is written.
            std::thread::sleep(std::time::Duration::from_millis(2100));
            assert_eq!(run(&env, &["status"]).0, Some(0));
            let cache = env.root().join(".opalro/cache/files");
            let old = std::fs::read(&cache).unwrap_or_default();
            env.overwrite("a.txt", "one\nWORK\n");
            match variant {
                0 => {}
                1 => {
                    let _ = std::fs::remove_file(&cache);
                }
                2 => std::fs::write(&cache, &old[..old.len() / 2]).unwrap(),
                3 => {
                    let mut b = old.clone();
                    let n = b.len() - 1;
                    b[n] ^= 1;
                    std::fs::write(&cache, b).unwrap();
                }
                4 => {
                    let mut b = old.clone();
                    b[20] ^= 0xff;
                    std::fs::write(&cache, b).unwrap();
                }
                5 => {
                    assert_eq!(run(&env, &["status"]).0, Some(0));
                }
                _ => {
                    // The cache from before the edit put back.
                    std::fs::write(&cache, &old).unwrap();
                }
            }
            let target = if cmd == "goto" { "side" } else { "side" };
            let (c, o, e) = run(&env, &[cmd, target]);
            let now = std::fs::read_to_string(env.root().join("a.txt")).unwrap_or_default();
            eprintln!("variant {variant} {cmd}: {c:?} a.txt {now:?} | {}", e.lines().next().unwrap_or(o.lines().next().unwrap_or("")));
            if !now.contains("WORK") {
                // Somewhere else? Shelves and safety copies.
                let all = support::entries(env.root());
                let kept = all.values().any(|(_, b, _)| String::from_utf8_lossy(b).contains("WORK"));
                eprintln!("   WORK not in a.txt; elsewhere in the folder: {kept}");
            }
        }
    }
}

/// A conflict below a folder the other line moved, both ways round; then resolve each way and
/// `who`.
#[test]
fn rt24_conflict_under_moved_folder() {
    for (mover, keep) in [("side", "mine"), ("side", "theirs"), ("main", "mine"), ("main", "theirs")] {
        let env = Env::empty();
        assert_eq!(run(&env, &["start", "--preset", "custom"]).0, Some(0));
        std::fs::create_dir_all(env.root().join("d")).unwrap();
        env.overwrite("d/x.txt", "one\ntwo\nthree\n");
        env.overwrite("d/y.txt", "why\n");
        assert_eq!(run(&env, &["save", "First"]).0, Some(0));
        assert_eq!(run(&env, &["line", "new", "side"]).0, Some(0));
        let edit = |who: &str| {
            if who == mover {
                std::fs::rename(env.root().join("d"), env.root().join("e")).unwrap();
                env.overwrite("e/x.txt", &format!("one\n{who}\nthree\n"));
            } else {
                env.overwrite("d/x.txt", &format!("one\n{who}\nthree\n"));
            }
        };
        edit("side");
        assert_eq!(run(&env, &["save", "Side"]).0, Some(0));
        assert_eq!(run(&env, &["goto", "main"]).0, Some(0));
        edit("main");
        assert_eq!(run(&env, &["save", "Main"]).0, Some(0));
        let (c, o, e) = run(&env, &["combine", "side"]);
        let (_, st, _) = run(&env, &["status"]);
        let files = support::entries(env.root());
        let names: Vec<_> = files.keys().filter(|k| !k.starts_with(".opalro")).collect();
        eprintln!("mover {mover}: combine {c:?}\n{o}{e}status:\n{st}folder {names:?}");
        let (c2, o2, e2) = run(&env, &["resolve", "--keep", keep, "d/x.txt"]);
        let (_, st2, _) = run(&env, &["status"]);
        let files = support::entries(env.root());
        let names: Vec<_> = files.keys().filter(|k| !k.starts_with(".opalro")).collect();
        eprintln!("resolve --keep {keep}: {c2:?} {o2}{e2}status:\n{st2}folder {names:?}");
        for f in ["d/x.txt", "e/x.txt"] {
            if env.root().join(f).exists() {
                let (cw, ow, ew) = run(&env, &["who", f]);
                eprintln!("who {f}: {cw:?}\n{ow}{ew}");
            }
        }
        let (vc, _, ve) = run(&env, &["verify"]);
        assert_eq!(vc, Some(0), "{ve}");
    }
}

/// Three common starting points: main and side each combine three lines, in other orders; then
/// side into main. Many rounds, so the starts' IDs come in every order.
#[test]
fn rt24_three_starts() {
    let mut seen = BTreeMap::new();
    for round in 0..12 {
        let env = Env::empty();
        assert_eq!(run(&env, &["start", "--preset", "custom"]).0, Some(0));
        env.overwrite("f.txt", "a\nb\nc\nd\ne\nf\n");
        assert_eq!(run(&env, &["save", "Base"]).0, Some(0));
        for (i, l) in ["one", "bee", "six"].iter().enumerate() {
            assert_eq!(run(&env, &["goto", "main"]).0, Some(0));
            assert_eq!(run(&env, &["line", "new", l]).0, Some(0));
            let mut ls = vec!["a", "b", "c", "d", "e", "f"];
            let t = format!("{l}{round}");
            ls[i * 2] = &t;
            env.overwrite("f.txt", &(ls.join("\n") + "\n"));
            if round % 3 == 0 {
                env.overwrite("g.txt", &format!("{l}\n"));
            }
            assert_eq!(run(&env, &["save", l]).0, Some(0));
        }
        assert_eq!(run(&env, &["goto", "main"]).0, Some(0));
        assert_eq!(run(&env, &["line", "new", "side"]).0, Some(0));
        env.overwrite("s.txt", "side\n");
        assert_eq!(run(&env, &["save", "S"]).0, Some(0));
        let orders = [["one", "bee", "six"], ["six", "one", "bee"]];
        for (line, order) in [("side", orders[1]), ("main", orders[0])] {
            assert_eq!(run(&env, &["goto", line]).0, Some(0));
            if line == "main" {
                env.overwrite("m.txt", "main\n");
                assert_eq!(run(&env, &["save", "M"]).0, Some(0));
            }
            for l in order {
                let (c, o, e) = run(&env, &["combine", l]);
                if o.contains("conflict") || e.contains("conflict") || c != Some(0) {
                    for f in ["f.txt", "g.txt"] {
                        let _ = run(&env, &["resolve", "--keep", "mine", f]);
                    }
                    let (_, st, _) = run(&env, &["status"]);
                    assert!(!st.contains("In conflict"), "{st}");
                }
            }
        }
        let (c, o, e) = run(&env, &["combine", "side"]);
        let first = format!("{c:?} {}", o.lines().next().unwrap_or(e.lines().next().unwrap_or("")));
        let first = first.split('"').next().unwrap_or("").to_owned();
        let (vc, _, ve) = run(&env, &["verify"]);
        assert_eq!(vc, Some(0), "{ve}");
        let f = std::fs::read_to_string(env.root().join("f.txt")).unwrap_or_default();
        eprintln!("round {round}: {c:?} {o}{e} f.txt {f:?}");
        *seen.entry(first).or_insert(0) += 1;
    }
    eprintln!("{seen:#?}");
}

/// A line whose history has a save the store lacks, beside one that holds the save: the amend
/// must be refused, and say what.
#[test]
fn rt24_amend_missing_save() {
    for holder_first in [false, true] {
        let env = Env::empty();
        assert_eq!(run(&env, &["start", "--preset", "custom"]).0, Some(0));
        env.overwrite("a.txt", "a\n");
        assert_eq!(run(&env, &["save", "A"]).0, Some(0));
        let a = main_tip(&env);
        let mut repo = Repository::open(env.root()).unwrap();
        let lock = repo.lock_for_change("save", &mut |_| {}).unwrap();
        let (head, entry) = repo.current().unwrap();
        let made_on = repo.store().save(&a, false).unwrap();
        let mut pack = repo.new_packs().unwrap();
        let mut planted = |parents: Vec<Id>, t: i64| {
            let mut author = made_on.author.clone();
            author.time_ms = t;
            let save = Save {
                root: made_on.root,
                parents,
                author: author.clone(),
                saver: author,
                message: "Planted".into(),
                change_id: ChangeId::random().unwrap(),
                move_notes: vec![],
                signature: vec![],
                extra: made_on.extra.clone(),
            };
            pack.add_record(ObjectKind::Save, &save.encode()).unwrap()
        };
        let broken = planted(vec![Id([7; 32])], 9_999_999_999_999);
        let holds = planted(vec![a], 1);
        let mut state = entry.state.clone();
        let (n1, n2) = if holder_first { ("ab", "cd") } else { ("cd", "ab") };
        state.lines.insert(n1.into(), holds);
        state.lines.insert(n2.into(), broken);
        let diary = DiaryEntry {
            parent: Some(head),
            action: Action::Command,
            command: "save".into(),
            who: entry.who.clone(),
            opal_version: "0.1.0".into(),
            state,
            folder_change: None,
        };
        repo.commit(&lock, pack, &diary).unwrap();
        drop(lock);
        drop(repo);
        env.overwrite("a.txt", "b\n");
        let (c, o, e) = run(&env, &["save", "--amend", "Amended"]);
        eprintln!("holder first {holder_first}: {c:?} {o}{e}");
    }
}

/// A random history of planted saves with lying clocks, lines, tags and server lines at random
/// saves, `main` at one of them; then `save --amend` must be refused exactly when another holds it.
#[test]
fn rt24_amend_holders_random() {
    let mut wrong = Vec::new();
    for seed in 1..=150u64 {
        let mut rng = Rng(0x9e37_79b9_7f4a_7c15 ^ seed.wrapping_mul(0x1234_5678_9abc));
        let env = Env::empty();
        env.opalro()
            .args(["start", "--preset", "custom"])
            .assert()
            .success();
        let first = main_tip(&env);
        let mut repo = Repository::open(env.root()).unwrap();
        let lock = repo.lock_for_change("save", &mut |_| {}).unwrap();
        let (head, entry) = repo.current().unwrap();
        let made_on = repo.store().save(&first, false).unwrap();
        let mut pack = repo.new_packs().unwrap();
        let mut saves: Vec<(Id, Vec<Id>)> = vec![(first, made_on.parents.clone())];
        let n = 6 + rng.below(30);
        for _ in 0..n {
            let mut parents = Vec::new();
            for _ in 0..1 + rng.below(3) {
                let p = saves[rng.below(saves.len())].0;
                if !parents.contains(&p) {
                    parents.push(p);
                }
            }
            let mut author = made_on.author.clone();
            author.time_ms = match rng.below(4) {
                0 => 1000,
                1 => -5000 + rng.below(10_000) as i64,
                _ => 1_700_000_000_000 + rng.below(1_000_000) as i64,
            };
            let save = Save {
                root: made_on.root,
                parents: parents.clone(),
                author: author.clone(),
                saver: author,
                message: "Planted".into(),
                change_id: ChangeId::random().unwrap(),
                move_notes: vec![],
                signature: vec![],
                extra: made_on.extra.clone(),
            };
            let id = pack.add_record(ObjectKind::Save, &save.encode()).unwrap();
            saves.push((id, parents));
        }
        let target = saves[1 + rng.below(n)].0;
        let mut state = entry.state.clone();
        state.lines.insert("main".into(), target);
        let mut holders: Vec<(String, Id)> = Vec::new();
        for i in 0..rng.below(6) {
            let at = saves[rng.below(saves.len())].0;
            state.lines.insert(format!("ab{i}"), at);
            holders.push((format!("line ab{i}"), at));
        }
        let mut tags = TagList::default();
        for i in 0..rng.below(3) {
            let at = saves[rng.below(saves.len())].0;
            tags.tags.insert(
                format!("v{i}.0"),
                Tag {
                    save: at,
                    message: String::new(),
                    git_tags: vec![],
                },
            );
            holders.push((format!("tag v{i}.0"), at));
        }
        if !tags.tags.is_empty() {
            state.tags = Some(pack.add_record(ObjectKind::TagList, &tags.encode()).unwrap());
        }
        let mut servers = ServerLines::default();
        for i in 0..rng.below(3) {
            let at = saves[rng.below(saves.len())].0;
            servers
                .servers
                .entry("origin".into())
                .or_default()
                .insert(format!("sb{i}"), at);
            holders.push((format!("server line sb{i}@origin"), at));
        }
        if !servers.servers.is_empty() {
            state.server_lines = Some(
                pack.add_record(ObjectKind::ServerLines, &servers.encode())
                    .unwrap(),
            );
        }
        let diary = DiaryEntry {
            parent: Some(head),
            action: Action::Command,
            command: "save".into(),
            who: entry.who.clone(),
            opal_version: "0.1.0".into(),
            state,
            folder_change: None,
        };
        repo.commit(&lock, pack, &diary).unwrap();
        drop(lock);
        drop(repo);
        let expect: Vec<&String> = holders
            .iter()
            .filter(|(_, at)| reaches(&saves, *at, target))
            .map(|(n, _)| n)
            .collect();
        std::fs::write(env.root().join("x.txt"), format!("{seed}\n")).unwrap();
        let out = env
            .opalro()
            .args(["save", "--amend", "Amended"])
            .output()
            .unwrap();
        let said = text(&out.stderr);
        let refused = out.status.code() != Some(0);
        let ok = if expect.is_empty() {
            !refused
        } else {
            let first = said.lines().next().unwrap_or("");
            let named = first
                .split_once("amend this save: ")
                .and_then(|(_, r)| r.split_once(" show"))
                .map_or("", |(n, _)| n);
            let (listed, more) = match named.rsplit_once(" and ") {
                Some((l, m)) if m.ends_with(" more") => {
                    (l.to_owned(), m.trim_end_matches(" more").parse().unwrap_or(0))
                }
                Some((l, m)) => (format!("{l}, {m}"), 0usize),
                None => (named.to_owned(), 0),
            };
            let listed: Vec<&str> = listed.split(", ").collect();
            refused
                && listed.len() + more == expect.len()
                && listed.iter().all(|l| expect.iter().any(|e| e == l))
        };
        if !ok {
            wrong.push(format!(
                "seed {seed}: expected holders {expect:?}, code {:?}: {said}",
                out.status.code()
            ));
        }
        eprintln!(
            "seed {seed}: {} saves, {} holders, expected {}, refused {refused}",
            n,
            holders.len(),
            expect.len()
        );
    }
    assert!(wrong.is_empty(), "{wrong:#?}");
}
