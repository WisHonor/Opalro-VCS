#!/usr/bin/env python3
"""Writes `count` packs of one small blob each into <objects>/pack, with their v2 indexes.
Usage: fillpacks.py <objects-dir> <count>"""
import hashlib, os, struct, sys, zlib

objects, count = sys.argv[1], int(sys.argv[2])
folder = os.path.join(objects, "pack")
os.makedirs(folder, exist_ok=True)
for i in range(count):
    body = b"filler %d\n" % i
    oid = hashlib.sha1(b"blob %d\0" % len(body) + body).digest()
    size = len(body)
    first = (3 << 4) | (size & 0x0F)
    size >>= 4
    header = bytearray()
    while size:
        header.append(first | 0x80)
        first = size & 0x7F
        size >>= 7
    header.append(first)
    entry = bytes(header) + zlib.compress(body)
    pack = b"PACK" + struct.pack(">II", 2, 1) + entry
    pack_sum = hashlib.sha1(pack).digest()
    pack += pack_sum
    fan = [0] * 256
    for b in range(oid[0], 256):
        fan[b] = 1
    idx = b"\xfftOc" + struct.pack(">I", 2) + b"".join(struct.pack(">I", f) for f in fan)
    idx += oid + struct.pack(">I", zlib.crc32(entry) & 0xFFFFFFFF) + struct.pack(">I", 12)
    idx += pack_sum
    idx += hashlib.sha1(idx).digest()
    name = "pack-" + pack_sum.hex()
    with open(os.path.join(folder, name + ".pack"), "wb") as f:
        f.write(pack)
    with open(os.path.join(folder, name + ".idx"), "wb") as f:
        f.write(idx)
