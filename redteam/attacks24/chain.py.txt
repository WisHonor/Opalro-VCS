#!/usr/bin/env python3
"""A fast-import stream: a file `depth` folders deep whose contents flip between two
versions in each of `commits` commits, so every commit's comparison reads two chains of
one-entry trees. Usage: chain.py depth commits [name]"""
import sys

depth = int(sys.argv[1])
commits = int(sys.argv[2])
name = sys.argv[3] if len(sys.argv) > 3 else "a"
path = "/".join([name] * depth + ["f"])
out = sys.stdout.buffer
out.write(b"blob\nmark :1\ndata 2\nX\n")
out.write(b"blob\nmark :2\ndata 2\nY\n")
for i in range(commits):
    msg = b"c%d\n" % i
    out.write(b"commit refs/heads/main\n")
    out.write(b"committer T <t@e> %d +0000\n" % (1000000 + i))
    out.write(b"data %d\n%s" % (len(msg), msg))
    out.write(b"M 100644 :%d %s\n" % (1 + (i % 2), path.encode()))
    out.write(b"\n")
out.write(b"done\n")
